Процедура ВыполнитьСинхронизацию() Экспорт

	// 1. Получаем список необработанных заказов.
	МассивЗаказов = ПолучитьНовыеЗаказы();
	Если МассивЗаказов = Неопределено Тогда
		Возврат;
	КонецЕсли;

	Если МассивЗаказов.Количество() = 0 Тогда
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Информация,
			, ,
			"Нет новых заказов."
		);
		Возврат;
	КонецЕсли;

	ЗаписьЖурналаРегистрации(
		"СинхрЗаказовСайта",
		УровеньЖурналаРегистрации.Информация,
		, ,
		"Получено заказов: " + Строка(МассивЗаказов.Количество())
	);

	// 2. Обрабатываем каждый заказ.
	Для Каждого Заказ Из МассивЗаказов Цикл

		СтрокаКлюча = Неопределено;
		Попытка
			СтрокаКлюча = НайтиИлиСоздатьКлючЗаказа(Заказ);
			Если СтрокаКлюча = Неопределено Тогда
				Продолжить;
			КонецЕсли;

			Если СтрокаКлюча.СтатусОбработки = "processed" Тогда
				Продолжить;
			КонецЕсли;

			// Создаем документ.
			Результат = СоздатьЗаказПокупателяИзЗаказа(Заказ);
			СсылкаДокумента = Результат["Ссылка"];
			НомерДокумента = Результат["Номер"];

			// Отправляем ack на сайт.
			РезультатAck = ОтправитьПодтверждениеЗаказа(
				Заказ["id"],
				СсылкаДокумента.УникальныйИдентификатор(),
				НомерДокумента,
				"Заказ создан в 1С"
			);

			// Фиксируем статус.
			СтрокаКлюча.СтатусОбработки = "processed";
			СтрокаКлюча.СсылкаЗаказаПокупателя = СсылкаДокумента;
			СтрокаКлюча.КомментарийОшибки = "";
			Объект = СтрокаКлюча.ПолучитьОбъект();
			Объект.Записать();

		Исключение
			Сообщение = "Заказ=" + Строка(Заказ["id"]) + " Ошибка=" + ОписаниеОшибки();
			ЗаписьЖурналаРегистрации(
				"СинхрЗаказовСайта",
				УровеньЖурналаРегистрации.Ошибка,
				, ,
				Сообщение
			);

			Если СтрокаКлюча <> Неопределено Тогда
				Объект = СтрокаКлюча.ПолучитьОбъект();
				Объект.СтатусОбработки = "error";
				Объект.КомментарийОшибки = Лев(Сообщение, 1000);
				Попытка
					Объект.Записать();
				Исключение
				КонецПопытки;
			КонецЕсли;
		КонецПопытки;

	КонецЦикла;

Функция ПолучитьНовыеЗаказы()

	ПараметрыСтроки = "?limit=" + Строка(МаксимальноеКоличествоЗаказовЗаРаз())
		+ "&format=json&status=received";

	Ответ = ВыполнитьHTTPЗапрос(
		"GET",
		"/api/v1/orders" + ПараметрыСтроки,
		Неопределено,
		True
	);

	Если Ответ = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если Ответ.КодСостояния < 200 Или Ответ.КодСостояния >= 300 Тогда
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Ошибка,
			, ,
			"GET заказов вернул HTTP " + Строка(Ответ.КодСостояния)
		);
		Возврат Неопределено;
	КонецЕсли;

	ТелоОтвета = СимволыHTMLВВнутреннююКодировку(Ответ.ПолучитьКакСтроку());

	Попытка
		ЧтениеJSON = Новый ЧтениеJSON();
		ЧтениеJSON.УстановитьСтроку(ТелоОтвета);
		Данные = ПрочитатьJSON(ЧтениеJSON, Истина);
	Исключение
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Ошибка,
			, ,
			"Не удалось распарсить JSON ответа GET: " + ОписаниеОшибки()
		);
		Возврат Неопределено;
	КонецПопытки;

	// Формат ответа:
	// {
	//   "status": "ok",
	//   "count": N,
	//   "orders": [ { id, status, fullName, phone, email, totalAmount, itemsCount,
	//                 deliveryMethod, comment, createdAt,
	//                 items, orderNumber, address, previewImage,
	//                 rawPayload: { items, promoCode, loyaltyCardNumber, giftCardNumber, customer, ... } } ]
	// }

	Если НЕ Данные.Свойство("orders") Тогда
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Информация,
			, ,
			"Ответ не содержит orders."
		);
		Возврат Новый Массив;
	КонецЕсли;

	Результат = Новый Массив;
	Для Каждого ЭлементЗаказа Из Данные["orders"] Цикл
		Результат.Добавить(ЭлементЗаказа);
	КонецЦикла;

	Возврат Результат;

Функция ОтправитьПодтверждениеЗаказа(
	IdЗаказа,
	УникальныйИдентификаторДокумента,
	НомерДокумента,
	Примечание)

	ТелоЗапроса = Новый Структура;
	ТелоЗапроса.Вставить("status", "processed");
	ТелоЗапроса.Вставить("onecDocId", Строка(УникальныйИдентификаторДокумента));
	ТелоЗапроса.Вставить("onecDocNumber", Строка(НомерДокумента));
	ТелоЗапроса.Вставить("note", Примечание);

	ТелоJSON = Новый ЗаписьJSON;
	ТелоJSON.УстановитьСтроку(Новый Соответствие);
	ЗаписатьJSON(ТелоJSON, ТелоЗапроса, Истина);
	СтрокаJSON = ТелоJSON.Закрыть();

	Ответ = ВыполнитьHTTPЗапрос(
		"PATCH",
		"/api/v1/orders/" + Строка(IdЗаказа) + "/ack",
		СтрокаJSON,
		Ложь
	);

	Если Ответ = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Если Ответ.КодСостояния < 200 Или Ответ.КодСостояния >= 300 Тогда
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Ошибка,
			, ,
			"PATCH ack вернул HTTP " + Строка(Ответ.КодСостояния)
				+ ". Тело=" + Лев(Ответ.ПолучитьКакСтроку(), 400)
		);
		Возврат Ложь;
	КонецЕсли;

	Возврат Истина;

////////////////////////////////////////////////////////////////////////////////
// Общий HTTP-клиент (умеет GET и PATCH)
////////////////////////////////////////////////////////////////////////////////

Функция ВыполнитьHTTPЗапрос(
	Метод,
	ОтносительныйПуть,
	ТелоЗапроса,
	ИспользоватьЗаголовокAcceptJSON)

	Попытка
		Подключение = Новый HTTPConnection;
		URL = Новый УникальныйURL(БазовыйURL());

		Порт = URL.Порт();
		Если Порт = 0 Тогда
			Если НСтар(URL.Схема(), "https") = 0 Тогда
				Порт = 443;
			Иначе
				Порт = 80;
			КонецЕсли;
		КонецЕсли;

		Подключение.Открыть(
			НСтар(URL.Схема(), "https") = 0,
			URL.Хост(),
			Порт,
			, ,
			Неопределено,
			Неопределено,
			ТаймаутHTTP()
		);

		Запрос = Новый HTTPЗапрос(ОтносительныйПуть);

		Если ИспользоватьЗаголовокAcceptJSON Тогда
			Запрос.Заголовки.Вставить("Accept", "application/json");
		КонецЕсли;

		Если ТелоЗапроса <> Неопределено И Метод <> "GET" Тогда
			ТелоКачествеМассива = ПолучитьБайтыИзСтрокиUTF8(ТелоЗапроса);
			Запрос.УстановитьТелоИзМассиваБайтов(ТелоКачествеМассива);
			Запрос.Заголовки.Вставить("Content-Type", "application/json");
		КонецЕсли;

		Если Метод = "GET" Тогда
			Ответ = Подключение.Get(Запрос);
		ИначеЕсли Метод = "PATCH" Тогда
			Ответ = Подключение.Patch(Запрос);
		ИначеЕсли Метод = "POST" Тогда
			Ответ = Подключение.Post(Запрос);
		ИначеЕсли Метод = "PUT" Тогда
			Ответ = Подключение.Put(Запрос);
		ИначеЕсли Метод = "DELETE" Тогда
			Ответ = Подключение.Delete(Запрос);
		Иначе
			Возврат Неопределено;
		КонецЕсли;

		Возврат Ответ;

	Исключение
		ЗаписьЖурналаРегистрации(
			"СинхрЗаказовСайта",
			УровеньЖурналаРегистрации.Ошибка,
			, ,
			"HTTP "+Метод+" "+ОтносительныйПуть+" → "+ОписаниеОшибки()
		);
		Возврат Неопределено;
	КонецПопытки;

////////////////////////////////////////////////////////////////////////////////
// Работа со справочником ключей (антидублирование)
////////////////////////////////////////////////////////////////////////////////

Функция НайтиИлиСоздатьКлючЗаказа(Заказ)

	IDЗаказа = Заказ["id"];
	Если IDЗаказа = Неопределено ИЛИ TrimAll(Строка(IDЗаказа)) = "" Тогда
		Возврат Неопределено;
	КонецЕсли;

	Найден = Справочники.КлючИдентификатораЗаказовСайта.НайтиПоКоду(Строка(IDЗаказа));
	Если Найден <> Неопределено Тогда
		Возврат Найден;
	КонецЕсли;

	Объект = Справочники.КлючИдентификатораЗаказовСайта.СоздатьЭлемент();
	Объект.Код = Строка(IDЗаказа);
	Объект.Наименование = Строка(IDЗаказа);

	Объект.СтатусОбработки = "new";
	Объект.КомментарийОшибки = "";

	Если Заказ.Свойство("totalAmount") Тогда
		Объект.СуммаЗаказа = Число(Заказ["totalAmount"]);
	КонецЕсли;

	Если Заказ.Свойство("createdAt") Тогда
		ДатаЗаказа = ПреобразоватьДатуИзISO8601(Заказ["createdAt"]);
		Если ДатаЗаказа <> Неопределено Тогда
			Объект.ДатаЗаказа = ДатаЗаказа;
		КонецЕсли;
	КонецЕсли;

	Попытка
		Объект.Записать();
	Исключение
		// Race condition: кто-то успел записать раньше.
		Найден = Справочники.КлючИдентификатораЗаказовСайта.НайтиПоКоду(Строка(IDЗаказа));
		Если Найден <> Неопределено Тогда
			Возврат Найден;
		КонецЕсли;
		ВызватьИсключение;
	КонецПопытки;

	Возврат Объект.Ссылка;

////////////////////////////////////////////////////////////////////////////////
// Создание ЗаказаПокупателя в 1С
////////////////////////////////////////////////////////////////////////////////

Функция СоздатьЗаказПокупателяИзЗаказа(Заказ)

	// Достаем товары.
	ТоварыЗаказа = ПолучитьТоварыЗаказа(Заказ);

	// Клиент (ищем/создаем по телефону).
	Контрагент = НайтиИлиСоздатьКонтрагента(Заказ);

	// Организация и договор. Пока берем умолчания.
	Организация = ПолучитьОрганизациюПоУмолчанию();
	Договор = ПолучитьДоговорПоУмолчанию(Организация, Контрагент);
	Склад = ПолучитьСкладПоУмолчанию();
	Валюта = ПолучитьВалютуПоУмолчанию();

	// Тип цены номенклатуры.
	ТипЦен = ПолучитьТипЦенПоУмолчанию();

	Документ = Документы.ЗаказПокупателя.Создать();
	Документ.Дата = ДатаДокументаИзЗаказа(Заказ);

	Документ.Контрагент = Контрагент;
	Документ.Организация = Организация;
	Если Договор <> Неопределено Тогда
		Документ.Договор = Договор;
	КонецЕсли;
	Документ.Склад = Склад;
	Документ.Валюта = Валюта;
	Документ.ВалютаДокумента = Валюта;
	Документ.ТипЦен = ТипЦен;
	Документ.Курс = 1;
	Документ.Кратность = 1;

	Если Заказ.Свойство("comment") И Заказ["comment"] <> Неопределено Тогда
		Коммент = "" + Заказ["comment"];
		Если Заказ.Свойство("deliveryMethod") И Заказ["deliveryMethod"] <> Неопределено Тогда
			Коммент = "Доставка: " + Заказ["deliveryMethod"] + ". " + Коммент;
		КонецЕсли;
		Если Заказ.Свойство("address") И Заказ["address"] <> Неопределено Тогда
			Коммент = "Адрес: " + Заказ["address"] + ". " + Коммент;
		КонецЕсли;
		Документ.Комментарий = Лев(Коммент, 250);
	КонецЕсли;

	// Заполняем табличную часть Товары.
	ТЧ = Документ.Товары;
	ТЧ.Очистить();

	СуммаПоДокументу = 0;
	КоличествоПоДокументу = 0;

	Для Каждого СтрокаТовара Из ТоварыЗаказа Цикл

		НоваяСтрока = ТЧ.Добавить();
		НоваяСтрока.Номенклатура = СтрокаТовара["Номенклатура"];
		НоваяСтрока.Количество = СтрокаТовара["Количество"];
		НоваяСтрока.Цена = СтрокаТовара["ЦенаЗаЕдиницу"];
		НоваяСтрока.Сумма = СтрокаТовара["ЦенаЗаЕдиницу"] * СтрокаТовара["Количество"];
		НоваяСтрока.НоменклатураСтрока = СтрокаТовара["Наименование"];
		НоваяСтрока.Скидка = 0;
		НоваяСтрока.СуммаСкидки = 0;

		СуммаПоДокументу = СуммаПоДокументу + НоваяСтрока.Сумма;
		КоличествоПоДокументу = КоличествоПоДокументу + СтрокаТовара["Количество"];

	КонецЦикла;

	Если Заказ.Свойство("totalAmount") Тогда
		ОжидаемаяСумма = Число(Заказ["totalAmount"]);
		Если ОжидаемаяСумма <> 0 И СуммаПоДокументу <> ОжидаемаяСумма Тогда
			Корректировка = ОжидаемаяСумма - СуммаПоДокументу;
			Если Абс(Корректировка) > 0.01 Тогда
				СтрокаДельты = ТЧ.Добавить();
				СтрокаДельты.Номенклатура = ПолучитьНоменклатуруКорректировки();
				СтрокаДельты.НоменклатураСтрока = "Разница между сайтом и номенклатурой";
				СтрокаДельты.Количество = 1;
				СтрокаДельты.Цена = Корректировка;
				СтрокаДельты.Сумма = Корректировка;
				СуммаПоДокументу = СуммаПоДокументу + Корректировка;
				КоличествоПоДокументу = КоличествоПоДокументу + 1;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	Документ.СуммаДокумента = СуммаПоДокументу;

	// Доставка.
	Если Заказ.Свойство("deliveryMethod") И Заказ["deliveryMethod"] <> Неопределено Тогда
		СпособДоставки = НайтиСпособДоставкиПоНаименованию(Заказ["deliveryMethod"]);
		Если СпособДоставки <> Неопределено И Документ.Свойство("СпособДоставки") Тогда
			Документ.СпособДоставки = СпособДоставки;
		КонецЕсли;
	КонецЕсли;

	Документ.Записать(РежимЗаписиДокумента.Запись);
	Документ.Провести();

	Результат = Новый Структура;
	Результат.Вставить("Ссылка", Документ.Ссылка);

Функция ПолучитьТоварыЗаказа(Заказ)

	МассивТоваров = Новый Массив;

	// 1. Первичный источник: top-level items, если есть.
	Если Заказ.Свойство("items") И ТипЗнч(Заказ["items"]) = Тип("Массив") Тогда
		МассивИсточник = Заказ["items"];
	ИначеЕсли Заказ.Свойство("rawPayload") И ТипЗнч(Заказ["rawPayload"]) = Тип("Структура") Тогда
		ЕСли Заказ["rawPayload"].Свойство("items") И ТипЗнч(Заказ["rawPayload"]["items"]) = Тип("Массив") Тогда
			МассивИсточник = Заказ["rawPayload"]["items"];
		Иначе
			МассивИсточник = Новый Массив;
		КонецЕсли;
	Иначе
		МассивИсточник = Новый Массив;
	КонецЕсли;

	Если МассивИсточник.Количество() = 0 Тогда
		// В целостности — если прилетел заказ без товаров, пишем одну заглушку.
		Заглушка = Новый Структура;
		Заглушка.Вставить("sku", Неопределено);
		Заглушка.Вставить("title", "Заказ без состава");
		Заглушка.Вставить("quantity", 1);
		Заглушка.Вставить("unitPrice", Заказ.Свойство("totalAmount") ? Число(Заказ["totalAmount"]) : 0);
		МассивИсточник.Добавить(Заглушка);
	КонецЕсли;

	Для Каждого ТоварИзJSON Из МассивИсточник Цикл

		Наименование = "(Без названия)";
		Если ТоварИзJSON.Свойство("title") И ТоварИзJSON["title"] <> Неопределено Тогда
			Наименование = "" + ТоварИзJSON["title"];
		КонецЕсли;

		Артикул = Неопределено;
		Если ТоварИзJSON.Свойство("sku") И ТоварИзJSON["sku"] <> Неопределено Тогда
			Артикул = "" + ТоварИзJSON["sku"];
		КонецЕсли;

		Количество = 1;
		Если ТоварИзJSON.Свойство("quantity") Тогда
			Количество = Число(ТоварИзJSON["quantity"]);
			Если Количество <= 0 Тогда
				Количество = 1;
			КонецЕсли;
		КонецЕсли;

		ЦенаЗаЕдиницу = 0;
		Если ТоварИзJSON.Свойство("unitPrice") И ТоварИзJSON["unitPrice"] <> Неопределено Тогда
			ЦенаЗаЕдиницу = Число(ТоварИзJSON["unitPrice"]);
		ИначеЕсли ТоварИзJSON.Свойство("price") И ТоварИзJSON["price"] <> Неопределено Тогда
			ЦенаЗаЕдиницу = Число(ТоварИзJSON["price"]);
		ИначеЕсли ТоварИзJSON.Свойство("sum") И ТоварИзJSON["sum"] <> Неопределено Тогда
			Если Количество > 0 Тогда
				ЦенаЗаЕдиницу = Число(ТоварИзJSON["sum"]) / Количество;
			КонецЕсли;
		КонецЕсли;

		Если Артикул <> Неопределено И TrimAll(Артикул) <> "" Тогда
			Номенклатура = НайтиНоменклатуруПоАртикулу(Артикул);
		Иначе
			Номенклатура = Неопределено;
		КонецЕсли;

		Если Номенклатура = Неопределено Тогда
			Номенклатура = ПолучитьНоменклатуруПоУмолчанию();
			ЗаписьЖурналаРегистрации(
				"СинхрЗаказовСайта",
				УровеньЖурналаРегистрации.Предупреждение,
				, ,
				"Не найдена номенклатура для SKU=" + Строка(Артикул)
					+ " Наименование=" + Наименование
			);
		КонецЕсли;

		СтрокаТовара = Новый Структура;
		СтрокаТовара.Вставить("Номенклатура", Номенклатура);
		СтрокаТовара.Вставить("Количество", Количество);
		СтрокаТовара.Вставить("ЦенаЗаЕдиницу", ЦенаЗаЕдиницу);
		СтрокаТовара.Вставить("Наименование", Лев(Наименование, 250));

		МассивТоваров.Добавить(СтрокаТовара);

	КонецЦикла;

	Возврат МассивТоваров;

Функция НайтиИлиСоздатьКонтрагента(Заказ)

	Телефон = Неопределено;
	Если Заказ.Свойство("phone") И Заказ["phone"] <> Неопределено Тогда
		Телефон = "" + Заказ["phone"];
	КонецЕсли;

	ФИО = "Покупатель";
	Если Заказ.Свойство("fullName") И Заказ["fullName"] <> Неопределено Тогда
		ФИО = "" + Заказ["fullName"];
	КонецЕсли;

	Email = Неопределено;
	Если Заказ.Свойство("email") И Заказ["email"] <> Неопределено Тогда
		Email = "" + Заказ["email"];
	КонецЕсли;

	// Ищем по телефону.
	Если Телефон <> Неопределено ИЛИ TrimAll(Телефон) <> "" Тогда
		Найден = НайтиКонтрагентаПоТелефону(Телефон);
		Если Найден <> Неопределено Тогда
			Возврат Найден;
		КонецЕсли;
	КонецЕсли;

	// Ищем по email.
	Если Email <> Неопределено И TrimAll(Email) <> "" Тогда
		Найден = НайтиКонтрагентаПоПочте(Email);
		Если Найден <> Неопределено Тогда
			Возврат Найден;
		КонецЕсли;
	КонецЕсли;

	// Создаем нового.
	Объект = Справочники.Контрагенты.СоздатьЭлемент();
	Объект.Имя = Лев(ФИО, 100);
	Объект.Наименование = Лев(ФИО, 100);
	Объект.ИНН = "";
	Объект.КПП = "";
	Объект.ЮридическоеФизическоеЛицо = ПеречисленияЮридическиеФизическиеЛица.ФизическоеЛицо;
	Объект.ТипКонтрагента = ПеречисленияТипыКонтрагентов.Покупатель;

	// Контактная информация.
	Если Телефон <> Неопределено Тогда
		Попытка
			Объект.КонтактнаяИнформация.Добавить(
				ПеречисленияВидыКонтактнойИнформации.Телефон,
				Телефон,
				ТекущаяДата()
			);
		Исключение
		КонецПопытки;
	КонецЕсли;

	Если Email <> Неопределено Тогда
		Попытка
			Объект.КонтактнаяИнформация.Добавить(
				ПеречисленияВидыКонтактнойИнформации.ЭлектроннаяПочта,
				Email,
				ТекущаяДата()
			);
		Исключение
		КонецПопытки;
	КонецЕсли;

	Попытка
		Объект.Записать();
	Исключение
		// Если не удалось создать новый — возвращаем дефолтного.
		Возврат ПолучитьКонтрагентаПоУмолчанию();
	КонецПопытки;

	Возврат Объект.Ссылка;

////////////////////////////////////////////////////////////////////////////////
// Работа с датами ISO 8601 (типа "2026-08-12T10:15:22Z")
////////////////////////////////////////////////////////////////////////////////

Функция ПреобразоватьДатуИзISO8601(Знач СтрокаДата)

	Если СтрокаДата = Неопределено ИЛИ ""+СтрокаДата = "" Тогда
		Возврат Неопределено;
	КонецЕсли;

	С = "" + СтрокаДата;

	// "2026-08-12T10:15:22Z"
	Стр1 = Заменить(С, "Z", "");
	Стр1 = Заменить(Стр1, "T", " ");

	Если Стр(Найти(Стр1, ".")) > 0 Тогда
		Стр1 = Лев(Стр1, Найти(Стр1, ".") - 1);
	КонецЕсли;
	Если Стр(Найти(Стр1, "+")) > 0 Тогда
		Стр1 = Лев(Стр1, Найти(Стр1, "+") - 1);
	КонецЕсли;

	Попытка
		ДатаUTC = Дата(Стр1);
		// Конвертируем UTC в локальное время сервера.
		Возврат ДатаUTC - РазницаСерверногоВремениИUTC();
	Исключение
		Возврат Неопределено;
	КонецПопытки;

КонецФункции

Функция РазницаСерверногоВремениИUTC()
	Возврат ТекущаяДата() - ТекущаяДатаUTC();
